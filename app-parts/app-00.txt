async function getLocation(force=false){
  const cached = store.get('dd_location');
  if(!force && cached && Date.now()-cached.savedAt < 30*60*1000){ state.location=cached; renderAll(); return cached; }
  if(!navigator.geolocation){ liveMessage('location unsupported'); throw new Error('Geolocation is not supported in this browser.'); }
  liveMessage('requesting location…');
  const pos = await new Promise((resolve,reject)=>navigator.geolocation.getCurrentPosition(resolve,reject,{enableHighAccuracy:true,timeout:12000,maximumAge:60000}));
  state.location = {lat:pos.coords.latitude, lon:pos.coords.longitude, accuracy:pos.coords.accuracy, savedAt:Date.now()};
  store.set('dd_location', state.location);
  liveMessage(`live · ±${Math.round(pos.coords.accuracy)}m`);
  await Promise.allSettled([loadWeather(true), loadPollen(true), primeDestinations()]);
  autoDetectSite();
  await refreshPulseRoutes();
  renderAll();
  return state.location;
}

function weatherCacheKey(){ if(!state.location)return null; return `dd_weather_${state.location.lat.toFixed(2)}_${state.location.lon.toFixed(2)}`; }
async function loadWeather(force=false){
  if(!state.location) return;
  const key=weatherCacheKey(); const cached=store.get(key);
  if(!force && cached && Date.now()-cached.savedAt < 10*60*1000){ state.weather=cached.data; renderWeather(); renderPulse(); renderSky(); renderPhoto(); return; }
  $('#weatherUpdated').textContent='Refreshing…';
  const {lat,lon}=state.location;
  const current='temperature_2m,apparent_temperature,relative_humidity_2m,precipitation,weather_code,cloud_cover,wind_speed_10m,is_day';
  const hourly='temperature_2m,precipitation_probability,weather_code,cloud_cover,visibility,dew_point_2m,wind_speed_10m,wind_gusts_10m,uv_index';
  const daily='weather_code,temperature_2m_max,temperature_2m_min,precipitation_probability_max,uv_index_max,sunrise,sunset,moonrise,moonset,moon_phase';
  const url=`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=${current}&hourly=${hourly}&daily=${daily}&temperature_unit=fahrenheit&wind_speed_unit=mph&precipitation_unit=inch&timezone=auto&forecast_days=16`;
  const r=await fetch(url); if(!r.ok) throw new Error(`Weather request failed (${r.status})`); const data=await r.json();
  state.weather=data; store.set(key,{savedAt:Date.now(),data});
  renderWeather(); renderPulse(); renderSky(); renderPhoto();
}

async function loadPollen(force=false){
  if(!state.location) return;
  const key=store.get('dd_pollen_key','');
  if(!key){ state.pollen=null; renderPollen(); return; }
  const cacheKey=`dd_pollen_${state.location.lat.toFixed(2)}_${state.location.lon.toFixed(2)}`; const cached=store.get(cacheKey);
  if(!force && cached && Date.now()-cached.savedAt < 60*60*1000){state.pollen=cached.data;renderPollen();return;}
  try{
    const u=`https://pollen.googleapis.com/v1/forecast:lookup?key=${encodeURIComponent(key)}&location.longitude=${state.location.lon}&location.latitude=${state.location.lat}&days=1&plantsDescription=false`;
    const r=await fetch(u); if(!r.ok) throw new Error('Pollen API'); const data=await r.json(); state.pollen=data; store.set(cacheKey,{savedAt:Date.now(),data});
  }catch(e){ state.pollen={error:true}; }
  renderPollen();
}
