async function primeDestinations(){
  const token=store.get('dd_mapbox_token',''); if(!token)return;
  for(const [key,d] of Object.entries(DESTINATIONS)){
    const address=store.get(`dd_${key}_address`,'');
    if(!address) continue;
    const old=store.get(`dd_dest_${key}`);
    if(old?.address===address && Number.isFinite(old.lat) && Number.isFinite(old.lon)) continue;
    try{ const u=`https://api.mapbox.com/search/geocode/v6/forward?q=${encodeURIComponent(address)}&country=US&types=address&limit=1&access_token=${encodeURIComponent(token)}`; const r=await fetch(u); if(!r.ok)continue; const j=await r.json(); const coords=j.features?.[0]?.geometry?.coordinates; if(coords)store.set(`dd_dest_${key}`,{lon:coords[0],lat:coords[1],address}); }catch{}
  }
}

async function routeRequest(origin,dest,departAt=null){
  const token=store.get('dd_mapbox_token',''); if(!token)throw new Error('Mapbox token not configured'); const at=departAt?`&depart_at=${encodeURIComponent(departAt)}`:''; const u=`https://api.mapbox.com/directions/v5/mapbox/driving-traffic/${origin.lon},${origin.lat};${dest.lon},${dest.lat}?alternatives=true&geometries=geojson&overview=full&steps=true&annotations=congestion_numeric,closure,duration,distance,speed${at}&access_token=${encodeURIComponent(token)}`; const r=await fetch(u); if(!r.ok)throw new Error(`Traffic request failed (${r.status})`); const j=await r.json(); if(j.code!=='Ok'||!j.routes?.length)throw new Error('No traffic route returned'); return j;
}
