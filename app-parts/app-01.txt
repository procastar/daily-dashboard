function currentHourIndex(){
  if(!state.weather?.hourly?.time) return -1; const now=Date.now(); let best=-1,bestDiff=Infinity; state.weather.hourly.time.forEach((t,i)=>{const d=Math.abs(new Date(t).getTime()-now);if(d<bestDiff){best=i;bestDiff=d}}); return best;
}
function todaySlice(){
  const w=state.weather; if(!w?.hourly?.time)return {idx:[],times:[]}; const now=new Date(); const day=now.toISOString().slice(0,10); const idx=[]; w.hourly.time.forEach((t,i)=>{if(t.startsWith(day))idx.push(i)}); return {idx,times:idx.map(i=>w.hourly.time[i])};
}
function getTodayPeak(arrName){

}
